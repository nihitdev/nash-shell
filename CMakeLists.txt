cmake_minimum_required(VERSION 3.20)
project(nash VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(NASH_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
find_package(tomlplusplus 3 REQUIRED)
add_library(nash_core
    src/util.cpp src/env.cpp src/lexer.cpp src/expand.cpp src/parser.cpp
    src/executor.cpp src/builtins.cpp src/prompt.cpp src/config.cpp
    src/editor.cpp src/completion.cpp src/highlight.cpp src/integrations.cpp src/shell.cpp)
target_include_directories(nash_core PUBLIC src)
target_link_libraries(nash_core PUBLIC tomlplusplus::tomlplusplus)
target_compile_options(nash_core PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(nash_core PUBLIC NASH_VERSION="${PROJECT_VERSION}")
include(GNUInstallDirs)
target_compile_definitions(nash_core PRIVATE NASH_THEME_DIR="${CMAKE_INSTALL_FULL_DATADIR}/nash/themes")
if(NASH_SANITIZE)
    target_compile_options(nash_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(nash_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(nash src/main.cpp)
target_link_libraries(nash PRIVATE nash_core)
install(TARGETS nash RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY themes/ DESTINATION ${CMAKE_INSTALL_DATADIR}/nash/themes)
install(FILES examples/config.nash DESTINATION ${CMAKE_INSTALL_DATADIR}/nash)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/nash)
install(FILES README.md CONTRIBUTING.md SECURITY.md CHANGELOG.md LICENSE DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DOCDIR}/docs FILES_MATCHING PATTERN "*.md" PATTERN "*.json")
install(FILES examples/config.nash examples/config.toml DESTINATION ${CMAKE_INSTALL_DOCDIR}/examples)
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_executable(nash_unit tests/unit.cpp)
    target_link_libraries(nash_unit PRIVATE nash_core)
    add_test(NAME unit COMMAND nash_unit)
    add_executable(nash_completion_tests tests/completion.cpp)
    target_link_libraries(nash_completion_tests PRIVATE nash_core)
    add_test(NAME completion COMMAND nash_completion_tests)
    add_test(NAME execution COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/execution.py $<TARGET_FILE:nash>)
    add_test(NAME interactive COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/interactive.py $<TARGET_FILE:nash>)
    add_test(NAME documentation COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/documentation.py $<TARGET_FILE:nash>)
    add_test(NAME reliability COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/reliability.py $<TARGET_FILE:nash>)
    set_tests_properties(execution interactive documentation reliability PROPERTIES TIMEOUT 90)
endif()
